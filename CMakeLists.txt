cmake_minimum_required(VERSION 3.20)
project(msx_music_render C)
add_executable(msx_render native/render.c vendor/emu2149/emu2149.c vendor/emu2413/emu2413.c vendor/emu2212/emu2212.c)

endif()

  target_link_libraries(msx_render m)
endif()
